```python
# Pipeline d'optimisation hiérarchique TransAs — 300 stops < 1.5 s sur Android
def optimize(stops, distance_matrix, start=0):
    """
    Renvoie l'ordre optimal des indices visités, start en position 0.

    Pipeline :
      n ≤ 40   → NN + 2-opt sur matrice NxN (cas direct)
      n > 40   → K-Means++ (Kmax=10, K≥3) → TSP centroids → NN+2-opt intra-grappe
    """
    if len(stops) <= 40:
        return nn_2opt(distance_matrix, start)        # sous-seconde

    K     = max(3, len(stops) // 30)                   # K = max(3, n/30) — pas de Kmax
    cents = kmeans_pp(stops, K=K)                       # K-means++ seeding
    order = tsp_centroids(cents, distance_matrix, start)
    return flat([
        nn_2opt(sub_matrix(c), 0) for c in order       # optimisation par grappe
    ])

# Invariants garantis (levée RuntimeError si violation — survit à `python -O`) :
if not all_indices_present(result, stops):
    raise RuntimeError(f"optimize: missing indices, got {result}")
if result[0] != start:
    raise RuntimeError(f"optimize: start not at position 0 (got {result[0]})")
# distance monotone : NN+2-opt ne peut pas être pire qu'un NN pur
```

### 🛰 Géocodage & Routage — fallback-first

```python
# core/geocoder.py — Nominatim avec cache JSON local + rate limit 1 req/s
class Geocoder:
    def __init__(self, cache_dir: Path, user_agent="TransAs/0.1"):
        self.cache = JsonCache(cache_dir / "geocode.json")   # JSON file on disk
        self.rate  = threading.Lock()                   # 1 req/sec strict (ToS)
        self.last  = 0.0

    def geocode(self, address: str) -> tuple[float, float] | None:
        if (cached := self.cache.get(address)):
            return cached                                 # offline-first hit
        with self.rate:                                   # bloque si < 1s depuis last
            wait = max(0, 1.0 - (time.time() - self.last))
            time.sleep(wait)
            self.last = time.time()
        coords = nominatim_request(address, user_agent=self.user_agent)
        self.cache.set(address, coords)
        return coords
```

```python
# core/router.py — OSRM /table avec fallback Haversine
def get_distance_matrix(coords, retries=3) -> list[list[float]]:
    """Si OSRM tombe, on retombe sur Haversine en local — jamais bloqué."""
    for attempt in range(retries):
        try:
            return osrm_table_request(coords, timeout=5 * (attempt + 1))
        except (URLError, Timeout) as e:
            log.warn(f"OSRM attempt {attempt+1}/{retries} failed: {e}")
    return haversine_matrix(coords)                        # offline fallback
```
